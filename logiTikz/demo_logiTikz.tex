\section{Additionneur complet (1 bit)}

$C_{out} = (A \cdot B) + (C_{in} \cdot (A \oplus B))$

\begin{center}
\begin{logigramme}[style=american,unit=1.5cm]{$S$}
  \INPUT{$A$}
  \INPUT{$B$}
  \INPUT{$C_{in}$}

  \GATE{x2}{xor}{x1}{i3}
  \OUT{x2}
\end{logigramme}
\quad
\begin{logigramme}[style=american,unit=1.5cm]{$C_{out}$}
  \INPUT{$A$}
  \INPUT{$B$}
  \INPUT{$C_{in}$}
  \GATE{x1}{xor}{i1}{i2}
  \GATE{a1}{and}{i1}{i2}
  \GATE{a2}{and}{x1}{i3}
  \GATE{or1}{or}{a1}{a2}
  \OUT{or1}
\end{logigramme}
\end{center}
