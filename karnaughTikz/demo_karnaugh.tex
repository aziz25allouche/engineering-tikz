%% ── 2 variables ─────────────────────────────────────────────
\section{2 variables --- $f(A,B)$}

\begin{center}
\begin{karnaugh}{2}{A,B}{1,0,0,1} % value filling order {(!A)(!B), (!A)B, A(!B), AB}
  \kvautogroup
\end{karnaugh}
\end{center}

$f = A \cdot B + \bar{A} \cdot \bar{B}$

%% ── 3 variables ─────────────────────────────────────────────
\section{3 variables --- $f(A,B,C)$}

\begin{center}
\begin{karnaugh}{3}{A,B,C}%
  {1,1,0,0, 1,X,0,X} % value filling order {(!A)(!B)(!C), (!A)(!B)C, ...}
  \kvautogroup
\end{karnaugh}
\end{center}

$f = \bar{B}$

%% ── 4 variables ─────────────────────────────────────────────
\section{4 variables --- $f(A,B,C,D)$}

  \kvautogroup

\end{center}

$f = A \cdot C + \bar{B} + C \cdot D$
